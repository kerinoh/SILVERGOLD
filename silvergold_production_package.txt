const express = require('express');
const http = require('http');
const { Server } = require('socket.io');
const crypto = require('crypto');

const app = express();
const server = http.createServer(app);
const io = new Server(server, { cors: { origin: "*" } });

app.use(express.json());

// System State Variables & Global Configurations
let userWallets = {
    "silvergold_player_1": 15000.00 // Seed player profile with KES 15,000.00
};

let silverGoldSettings = {
    minDeposit: 100,
    minStake: 10,
    minWithdrawal: 200,
    gatewayProvider: "Safaricom_Daraja_Production",
    adminPasscode: "kk1325"
};

let gameState = {
    status: "WAITING", // WAITING, FLYING, CRASHED
    currentMultiplier: 1.00,
    activeCrashPoint: 1.00,
    roundId: 8001,
    timeElapsed: 0,
    forcedOverrideMultiplier: null,
    activeRoundBets: { panelA: null, panelB: null },
    history: [
        { id: 7998, point: 1.45, tag: "blue" },
        { id: 7999, point: 5.20, tag: "purple" },
        { id: 8000, point: 78.40, tag: "gold" }
    ]
};

const MIN_CRASH = 1.00;
const MAX_CRASH = 80.00;

function generateNextCrashPoint() {
    if (gameState.forcedOverrideMultiplier) {
        const injectedPoint = gameState.forcedOverrideMultiplier;
        gameState.forcedOverrideMultiplier = null;
        return injectedPoint;
    }
    const serverSeed = crypto.randomBytes(32).toString('hex');
    const hashRatio = parseInt(serverSeed.substring(0, 13), 16) / 0xfffffffffffff;
    const biasedRatio = Math.pow(hashRatio, 0.75); // Skews curve to yield more 2.00+ multipliers (~60-65% odds)
    let computedPoint = MIN_CRASH + (biasedRatio * (MAX_CRASH - MIN_CRASH));
    
    if (parseInt(serverSeed.substring(13, 15), 16) % 33 === 0) return MIN_CRASH; // 3% House edge drop
    return parseFloat(computedPoint.toFixed(2));
}

function initiateFlightLifecycleLoop() {
    gameState.status = "WAITING";
    gameState.currentMultiplier = 1.00;
    gameState.timeElapsed = 0;
    gameState.activeCrashPoint = generateNextCrashPoint();
    gameState.activeRoundBets.panelA = null;
    gameState.activeRoundBets.panelB = null;

    io.emit("game_waiting", { roundId: gameState.roundId, history: gameState.history });

    setTimeout(() => {
        gameState.status = "FLYING";
        io.emit("game_started");

        const tickRateMs = 30;
        const flightTicker = setInterval(() => {
            gameState.timeElapsed += (tickRateMs / 1000);

            if (gameState.currentMultiplier < 30.00) {
                gameState.currentMultiplier = 1.00 + Math.pow(gameState.timeElapsed * 1.1, 1.8);
            } else {
                const zone2TimeOffset = gameState.timeElapsed - 24.2;
                gameState.currentMultiplier = 30.00 + (zone2TimeOffset * 3.16) + Math.pow(zone2TimeOffset * 1.4, 2.5);
            }

            if (gameState.currentMultiplier >= gameState.activeCrashPoint || gameState.currentMultiplier >= MAX_CRASH) {
                gameState.currentMultiplier = Math.min(gameState.activeCrashPoint, MAX_CRASH);
                clearInterval(flightTicker);
                triggerCrashResolutionSequence();
            } else {
                io.emit("multiplier_tick", { multiplier: parseFloat(gameState.currentMultiplier.toFixed(2)) });
            }
        }, tickRateMs);
    }, 5000);
}

function triggerCrashResolutionSequence() {
    gameState.status = "CRASHED";
    let statusColorTag = "blue";
    if (gameState.currentMultiplier >= 2.00) statusColorTag = "purple";
    if (gameState.currentMultiplier >= 10.00) statusColorTag = "gold";

    gameState.history.unshift({ id: gameState.roundId, point: gameState.currentMultiplier, tag: statusColorTag });
    if (gameState.history.length > 15) gameState.history.pop();

    io.emit("game_crashed", { finalMultiplier: gameState.currentMultiplier });
    gameState.roundId++;
    setTimeout(initiateFlightLifecycleLoop, 3000);
}

io.on("connection", (socket) => {
    socket.emit("wallet_sync", { currentBalanceKES: userWallets["silvergold_player_1"] });

    socket.on("submit_stake", (data) => {
        const { userId, targetPanel, stakeValueKES } = data;
        if (gameState.status !== "WAITING") return socket.emit("operation_error", { message: "Betting locked for this round!" });
        if (userWallets[userId] < stakeValueKES) return socket.emit("operation_error", { message: "Insufficient KES wallet balance!" });
        if (stakeValueKES < silverGoldSettings.minStake) return socket.emit("operation_error", { message: `Min stake is KES ${silverGoldSettings.minStake}` });

        userWallets[userId] -= stakeValueKES;
        gameState.activeRoundBets[targetPanel] = { userId, stakedAmount: stakeValueKES, status: "PLACED" };
        socket.emit("stake_acknowledged", { panel: targetPanel, revisedBalanceKES: userWallets[userId] });
        io.emit("lobby_roster_sync", gameState.activeRoundBets);
    });

    socket.on("trigger_cashout", (data) => {
        const { userId, targetPanel } = data;
        if (gameState.status !== "FLYING") return;
        
        const historicalRecord = gameState.activeRoundBets[targetPanel];
        if (!historicalRecord || historicalRecord.userId !== userId || historicalRecord.status !== "PLACED") return;

        const liveMultiplierSnapshot = parseFloat(gameState.currentMultiplier.toFixed(2));
        const finalWinningsPayoutKES = historicalRecord.stakedAmount * liveMultiplierSnapshot;

        userWallets[userId] += finalWinningsPayoutKES;
        historicalRecord.status = "CASHED_OUT";
        historicalRecord.settledWinnings = finalWinningsPayoutKES;
        historicalRecord.cashoutOdds = liveMultiplierSnapshot;

        socket.emit("cashout_confirmed", { 
            panel: targetPanel, 
            payoutKES: finalWinningsPayoutKES, 
            multiplier: liveMultiplierSnapshot,
            revisedBalanceKES: userWallets[userId]
        });
        io.emit("lobby_roster_sync", gameState.activeRoundBets);
    });
});

app.post('/api/v1/admin/inject-signal', (req, res) => {
    const { password, targetOdds } = req.body;
    if (password !== silverGoldSettings.adminPasscode) return res.status(403).json({ error: "Access Denied." });
    gameState.forcedOverrideMultiplier = Math.min(Math.max(parseFloat(targetOdds), MIN_CRASH), MAX_CRASH);
    res.status(200).json({ success: true, message: `Forced endpoint locked at ${gameState.forcedOverrideMultiplier}x` });
});

app.post('/api/v1/control-center/update-settings', (req, res) => {
    const { passcode, minDeposit, minStake, minWithdrawal, newPasscode } = req.body;
    if (passcode !== silverGoldSettings.adminPasscode) return res.status(403).json({ error: "Access Denied." });
    if (minDeposit) silverGoldSettings.minDeposit = Number(minDeposit);
    if (minStake) silverGoldSettings.minStake = Number(minStake);
    if (minWithdrawal) silverGoldSettings.minWithdrawal = Number(minWithdrawal);
    if (newPasscode) silverGoldSettings.adminPasscode = newPasscode;
    res.json({ success: true, message: "Settings updated successfully." });
});

server.listen(5000, () => {
    console.log("✈️ SilverGold Backend Running on Port 5000");
    initiateFlightLifecycleLoop();
});
